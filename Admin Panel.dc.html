2 · ขั้นตอนติดตั้งจริง

  1. สมัคร supabase.com (ฟรี) แล้วสร้าง Project ใหม่ เลือก region Singapore
  2. ไป SQL Editor วางสคริปต์ตารางด้านขวา แล้วกด Run
  3. ไป Storage สร้าง bucket ชื่อ school-media ตั้งเป็น Public
  4. ไป Authentication → Users เพิ่มอีเมลผู้ดูแล 1 บัญชี (ตามที่ตั้งไว้ในเมนู “บัญชีผู้ดูแล”)
  5. คัดลอก Project URL และ anon key จากหน้า Settings → API มาใส่ช่องด้านบน
  6. อัปโหลดโฟลเดอร์เว็บขึ้น Netlify (ลากไฟล์วางที่ app.netlify.com/drop ได้เลย) หรือ Vercel / GitHub Pages
  7. ผูกโดเมนของโรงเรียนที่หน้า Domain settings แล้วเปิด HTTPS อัตโนมัติ

ทั้ง Netlify, Vercel และ GitHub Pages เป็นโฮสต์ไฟล์แบบ static ที่ใช้ฟรีได้ ระบบล็อกอินและฐานข้อมูลอยู่ฝั่ง Supabase จึงไม่ต้องเช่าเซิร์ฟเวอร์เพิ่ม

ตอนนี้ระบบบันทึกข้อมูลไว้ในเครื่องของคุณ (บันทึกได้ ใช้งานได้จริง แต่ยังไม่แชร์ข้ามเครื่อง) เมื่อกรอกค่าเชื่อมต่อครบและอัปโหลดขึ้นโฮสต์แล้ว ข้อมูลจะย้ายไปอยู่บนฐานข้อมูลกลางทันที

สคริปต์สร้างตาราง (SQL)

-- ข้อมูลรายส่วนของหน้าเว็บ (key/value แบบยืดหยุ่น)
create table site_content (
  key         text primary key,     -- 'hero', 'about', 'contact'
  data        jsonb not null,
  updated_at  timestamptz default now(),
  updated_by  uuid references auth.users
);

-- รายการที่มีหลายแถว: ข่าว แบนเนอร์ บุคลากร
create table entries (
  id          uuid primary key default gen_random_uuid(),
  collection  text not null,        -- 'news' | 'banner' | 'staff'
  position    int  not null default 0,
  published   boolean default true,
  data        jsonb not null,
  created_at  timestamptz default now()
);
create index on entries (collection, position);

-- ไฟล์รูป/วิดีโอที่อัปโหลด
create table media (
  id         uuid primary key default gen_random_uuid(),
  path       text not null,
  kind       text not null,         -- 'image' | 'video' | 'file'
  title      text,
  created_at timestamptz default now()
);

-- สิทธิ์ผู้ดูแล
create table profiles (
  id    uuid primary key references auth.users on delete cascade,
  name  text,
  role  text not null default 'editor'  -- 'admin' | 'editor'
);

-- ทุกคนอ่านได้ เฉพาะผู้ล็อกอินแก้ไขได้
alter table site_content enable row level security;
alter table entries      enable row level security;
alter table media        enable row level security;

create policy "public read" on site_content for select using (true);
create policy "public read" on entries      for select using (true);
create policy "public read" on media        for select using (true);

create policy "admin write" on site_content for all
  using (auth.role() = 'authenticated')
  with check (auth.role() = 'authenticated');
create policy "admin write" on entries for all
  using (auth.role() = 'authenticated')
  with check (auth.role() = 'authenticated');
create policy "admin write" on media for all
  using (auth.role() = 'authenticated')
  with check (auth.role() = 'authenticated');

โค้ดเชื่อมหน้าเว็บกับฐานข้อมูล

import { createClient } from '@supabase/supabase-js'

const db = createClient(PROJECT_URL, ANON_KEY)

// อ่านเนื้อหามาแสดงบนหน้าเว็บ
const { data } = await db
  .from('site_content')
  .select('key, data')

// อ่านข่าวเรียงตามลำดับ
const { data: news } = await db
  .from('entries')
  .select('*')
  .eq('collection', 'news')
  .eq('published', true)
  .order('position')

// ล็อกอินผู้ดูแล
await db.auth.signInWithPassword({ email, password })

// บันทึกจากหน้า Admin
await db.from('site_content')
  .upsert({ key: 'hero', data: heroValues })

// อัปโหลดรูป
await db.storage.from('school-media')
  .upload(`news/${file.name}`, file)
