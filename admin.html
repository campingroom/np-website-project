2 · ขั้นตอนติดตั้งจริง

  1. เข้า supabase.com → เปิดโปรเจกต์ของโรงเรียน
  2. เมนู SQL Editor → New query → วางสคริปต์ด้านขวาทั้งหมด → กด Run (รันครั้งเดียวพอ)
  3. เมนู Storage → New bucket → ตั้งชื่อ school-media → ติ๊ก Public bucket
  4. เมนู Authentication → Users → Add user → ใส่อีเมลและรหัสผ่านของผู้ดูแล (ติ๊ก Auto Confirm User)
  5. เมนู Project Settings → API → คัดลอก Project URL และ anon public key มาวางในช่องด้านซ้าย แล้วกด “ทดสอบการเชื่อมต่อ”
  6. กด บันทึกการเปลี่ยนแปลง หนึ่งครั้ง เพื่ออัปโหลดเนื้อหาปัจจุบันขึ้นฐานข้อมูลเป็นครั้งแรก
  7. ออกจากระบบแล้วเข้าใหม่ด้วย อีเมล + รหัสผ่านจากขั้นตอนที่ 4 (ระบบจะเปลี่ยนไปใช้บัญชี Supabase อัตโนมัติ)
  8. อัปโหลดเว็บขึ้น Vercel แล้วเปิด Tweaks ของหน้าเว็บ กรอก Project URL / anon key ชุดเดียวกัน เพื่อให้ผู้เข้าชมทุกคนเห็นข้อมูลจากฐานข้อมูล

ทั้ง Netlify, Vercel และ GitHub Pages เป็นโฮสต์ไฟล์แบบ static ที่ใช้ฟรีได้ ระบบล็อกอินและฐานข้อมูลอยู่ฝั่ง Supabase จึงไม่ต้องเช่าเซิร์ฟเวอร์เพิ่ม

ตอนนี้ระบบบันทึกข้อมูลไว้ในเครื่องของคุณ (บันทึกได้ ใช้งานได้จริง แต่ยังไม่แชร์ข้ามเครื่อง) เมื่อกรอกค่าเชื่อมต่อครบและอัปโหลดขึ้นโฮสต์แล้ว ข้อมูลจะย้ายไปอยู่บนฐานข้อมูลกลางทันที

สคริปต์สร้างตาราง (SQL)

-- 1) ตารางเก็บเนื้อหาทั้งเว็บ (คีย์เดียว: 'cms')
create table if not exists site_content (
  key        text primary key,
  data       jsonb not null,
  updated_at timestamptz default now()
);

alter table site_content enable row level security;

-- ผู้เข้าชมเว็บ: อ่านได้อย่างเดียว
create policy "public read"  on site_content for select using (true);
-- ผู้ดูแลที่ล็อกอินแล้ว: เพิ่ม/แก้ได้
create policy "admin insert" on site_content for insert to authenticated with check (true);
create policy "admin update" on site_content for update to authenticated using (true);

-- 2) สิทธิ์ของ Storage bucket ชื่อ school-media
create policy "media public read" on storage.objects
  for select using (bucket_id = 'school-media');
create policy "media admin write" on storage.objects
  for insert to authenticated with check (bucket_id = 'school-media');
create policy "media admin update" on storage.objects
  for update to authenticated using (bucket_id = 'school-media');
create policy "media admin delete" on storage.objects
  for delete to authenticated using (bucket_id = 'school-media');

โครงสร้างข้อมูลที่ระบบใช้จริง

เนื้อหาทั้งเว็บถูกเก็บเป็นข้อมูลก้อนเดียวในแถวที่ key = 'cms' ของตาราง site_content ส่วนรูปภาพเก็บเป็นไฟล์ใน Storage bucket ชื่อ school-media โค้ดด้านล่างคือสิ่งที่ระบบทำให้อัตโนมัติ ไม่ต้องเขียนเพิ่ม

const db = createClient(PROJECT_URL, ANON_KEY)

// หน้าเว็บ: อ่านเนื้อหาทั้งหมด (แถวเดียว)
const { data } = await db
  .from('site_content')
  .select('data, updated_at')
  .eq('key', 'cms')
  .maybeSingle()

// Admin: ล็อกอินผู้ดูแล
await db.auth.signInWithPassword({ email, password })

// Admin: กดบันทึก = เขียนทับแถวเดิม
await db.from('site_content').upsert(
  { key: 'cms', data: allContent, updated_at: new Date().toISOString() },
  { onConflict: 'key' }
)

// Admin: อัปโหลดรูปเข้า Storage แล้วเก็บเฉพาะ URL
const { data: up } = await db.storage
  .from('school-media')
  .upload(path, file)
const url = db.storage.from('school-media')
  .getPublicUrl(up.path).data.publicUrl
